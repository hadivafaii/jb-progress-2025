Identify low contrast patches#

Bottom 5%

q = 0.05

vars = np.var(tonp(x), axis=(-1, -2))
vars = np.log(vars.ravel())
thres = np.quantile(vars, q)
accept = vars > thres

msg = f"# removed: {(~accept).sum()} ——— # final: {accept.sum()}"
print(msg)
