def pareto_front_indices(f1, f2):
    # maximize both f1 and f2
    f1, f2 = np.asarray(f1), np.asarray(f2)
    n = len(f1)
    is_dom = np.zeros(n, dtype=bool)
    for i in range(n):
        if is_dom[i]: 
            continue
        # j dominates i if j>=i in both and > in at least one
        dominate_i = (f1 >= f1[i]) & (f2 >= f2[i]) & ((f1 > f1[i]) | (f2 > f2[i]))
        if dominate_i.any():
            is_dom[i] = True
    return np.where(~is_dom)[0]

def crowding_distance(f1, f2):
    # larger distance means more isolated
    n = len(f1)
    dist = np.zeros(n)
    for f in (np.asarray(f1), np.asarray(f2)):
        order = np.argsort(f)
        dist[order[0]] = dist[order[-1]] = np.inf
        span = f[order[-1]] - f[order[0]]
        if span == 0:
            continue
        for i in range(1, n - 1):
            dist[order[i]] += (f[order[i + 1]] - f[order[i - 1]]) / span
    return dist

def select_by_pareto(r2, dtheta, k=6):
    r2 = np.asarray(r2)
    a = np.abs(np.asarray(dtheta)) / 4.0
    chosen = []
    remaining = np.arange(len(r2))
    while len(chosen) < k and len(remaining) > 0:
        front = pareto_front_indices(r2[remaining], a[remaining])
        front_idx = remaining[front]
        if len(chosen) + len(front_idx) <= k:
            chosen.extend(front_idx.tolist())
            remaining = np.setdiff1d(remaining, front_idx, assume_unique=False)
        else:
            # break ties using crowding distance to encourage spread
            cd = crowding_distance(r2[front_idx], a[front_idx])
            order = np.argsort(-cd)
            need = k - len(chosen)
            chosen.extend(front_idx[order[:need]].tolist())
            break
    return np.array(chosen)
