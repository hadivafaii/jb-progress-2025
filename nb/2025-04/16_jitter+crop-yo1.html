        tr.model.update_time(t)
        inpt = feeder[t]
        output = tr.model.xtract_ftr(inpt, return_extras=items)
        for k, d in output.items():
            r2[k][t] = d['r2'].mean(0).item()

        if (t + 1) == t_total:
